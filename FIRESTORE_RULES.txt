rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function role() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }
    function admin() { return signedIn() && role() == 'admin'; }
    function leaderOrAdmin() { return signedIn() && (role() == 'leader' || role() == 'admin'); }

    match /users/{userId} {
      allow create: if signedIn() && request.auth.uid == userId;
      allow read: if signedIn();
      allow update, delete: if admin();
    }
    match /workers/{id} { allow read: if signedIn(); allow write: if admin(); }
    match /sites/{id} { allow read: if signedIn(); allow write: if admin(); }
    match /attendance/{id} { allow read, create: if signedIn(); allow update, delete: if leaderOrAdmin(); }
    match /materials/{id} { allow read: if signedIn(); allow write: if leaderOrAdmin(); }
    match /profits/{id} { allow read, write: if admin(); }
    match /expenses/{id} { allow read, write: if admin(); }
    match /bills/{id} { allow read, write: if admin(); }
  }
}
